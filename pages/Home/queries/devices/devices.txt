WITH usage_agg AS (
  SELECT 
    device_id,
    COUNT(*) AS total_entries,
    ROUND(SUM(download_bytes) / 1073741824.0, 2) AS total_download_mb,
    ROUND(SUM(upload_bytes)   / 1073741824.0, 2) AS total_upload_mb,
    ROUND(SUM(download_bytes + upload_bytes) / 1073741824.0, 2) AS total_mb,
    ROUND(AVG(download_bytes) / 1048576.0, 2) AS avg_download_mb,
    ROUND(AVG(upload_bytes)   / 1048576.0, 2) AS avg_upload_mb
  FROM network_usage
  GROUP BY device_id
),

last_scan AS (
  SELECT DISTINCT ON (device_id)
    device_id,
    status,
    rtt_ms,
    scan_time
  FROM device_scan_log
  ORDER BY device_id, scan_time DESC
)

SELECT 
  d.id AS device_id,
  d.vendor,
  COALESCE(d.display_name, d.hostname, d.name, d.mac_address::text) AS name,
  d.hostname,
  d.mac_address,
  d.ip_address,
  d.is_dhcp,
  u.total_entries,
  u.total_download_mb,
  u.total_upload_mb,
  u.total_mb,
  u.avg_download_mb,
  u.avg_upload_mb,
  ls.status,
  ls.rtt_ms,
  ls.scan_time,
  (
    SELECT MAX(scan_time)
    FROM device_scan_log dsl2
    WHERE dsl2.device_id = d.id AND dsl2.status = true
  ) AS last_seen
FROM network_device d
LEFT JOIN usage_agg u ON u.device_id = d.id
LEFT JOIN last_scan ls ON ls.device_id = d.id
ORDER BY u.total_mb DESC NULLS LAST;
